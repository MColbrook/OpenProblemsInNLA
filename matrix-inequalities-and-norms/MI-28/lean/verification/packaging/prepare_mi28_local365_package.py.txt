retained = {}

def copy(src, rel, compress=False):
    src = Path(src)
    assert src.is_file() and not src.is_symlink(), src
    data = src.read_bytes()
    raw_sha = hashlib.sha256(data).hexdigest()
    encoding = 'identity'
    if compress and len(data) > 50000:
        rel += '.gz'
        data = gzip.compress(data, mtime=0)
        encoding = 'gzip'
    dst = P / rel
    dst.parent.mkdir(parents=True, exist_ok=True)
    if dst.exists():
        assert dst.read_bytes() == data, dst
    else:
        dst.write_bytes(data)
    retained[str(src)] = {'source': str(src), 'source_sha256': raw_sha, 'destination': rel,
        'sha256': sha(dst), 'encoding': encoding, 'decoded_sha256': raw_sha}
    return rel

def write(rel, obj):
    p = P / rel
    p.parent.mkdir(parents=True, exist_ok=True)
    p.write_text(json.dumps(obj, indent=2) + '\n')

def archive(src, rel):
    src = Path(src)
    dst = P / rel
    dst.parent.mkdir(parents=True, exist_ok=True)
    inventory = {}
    with dst.open('wb') as raw, gzip.GzipFile(fileobj=raw, mode='wb', mtime=0) as gz, tarfile.open(fileobj=gz, mode='w') as tar:
        for f in sorted(src.rglob('*')):
            assert not f.is_symlink(), f
            if not f.is_file():
                continue
            assert f.suffix not in {'.pdf', '.olean', '.ilean', '.o', '.so', '.a', '.zip'}, f
            data = f.read_bytes()
            name = str(f.relative_to(src))
            info = tarfile.TarInfo(name)
            info.size, info.mode = len(data), 0o644
            tar.addfile(info, io.BytesIO(data))
            inventory[name] = hashlib.sha256(data).hexdigest()
    return {'source_directory': str(src), 'archive': rel,
        'archive_sha256': sha(dst), 'payload_sha256': inventory}
