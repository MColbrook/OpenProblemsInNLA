def privacy(data, name):
    if name.endswith('.tar.gz'):
        with tarfile.open(fileobj=io.BytesIO(data), mode='r:gz') as tar:
            for m in tar.getmembers():
                assert m.isfile() and not PurePosixPath(m.name).is_absolute() and '..' not in PurePosixPath(m.name).parts
                privacy(tar.extractfile(m).read(), m.name)
    elif name.endswith('.gz'):
        privacy(gzip.decompress(data), name[:-3])
    else:
        assert not re.search(rb'[A-Za-z0-9_.+%\-]+@[A-Za-z0-9.\-]+\.[A-Za-z]{2,}', data), name
